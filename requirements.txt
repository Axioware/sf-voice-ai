# SF Voice AI — System Requirements
# This is a Node.js/Electron desktop app. npm handles JS dependencies (see package.json).
# The items below are system-level prerequisites.

# ── Runtime ──────────────────────────────────────────────────────────────────
Node.js >= 18.0.0        # LTS recommended; Electron 28 requires Node 18+
npm >= 9.0.0             # bundled with Node 18+

# ── Electron (installed via npm) ─────────────────────────────────────────────
electron >= 28.0.0       # see package.json devDependencies

# ── Audio (all platforms) ───────────────────────────────────────────────────
# Actual audio capture is done by services/audioCapture.js spawning `ffmpeg`
# directly (pulse on Linux, dshow on Windows, avfoundation on Mac).
# node-record-lpcm16 is listed in package.json but is NOT used by the app —
# it's a leftover dependency from an earlier implementation.
ffmpeg                   # required on every platform
                         # Install: sudo apt install ffmpeg      (Linux)
                         #      or: brew install ffmpeg           (Mac)
                         #      or: winget install ffmpeg         (Windows)

# ── Audio (Linux) ────────────────────────────────────────────────────────────
pulseaudio               # or pipewire-pulse — ffmpeg captures via `-f pulse`;
                         # `pactl` is also used to auto-detect the default
                         # sink's monitor (lead audio) and list devices

# ── Audio (macOS) ────────────────────────────────────────────────────────────
# BlackHole (https://github.com/ExistentialAudio/BlackHole) — free virtual
# audio device, needed to capture the other side of the call (not just mic)

# ── Audio (Windows) ──────────────────────────────────────────────────────────
# VB-Audio Virtual Cable (https://vb-audio.com/Cable) — free virtual audio
# device, needed to capture the other side of the call (not just mic)

# ── API Keys / config (set in .env — copy from .env.example) ────────────────
ANTHROPIC_API_KEY        # required — https://console.anthropic.com
DEEPGRAM_API_KEY         # required — https://console.deepgram.com
SF_INSTANCE_URL          # optional — enables Salesforce lead-by-phone lookup
SF_CLIENT_ID             # optional — Salesforce Connected App Consumer Key
SF_CLIENT_SECRET         # optional — Salesforce Connected App Consumer Secret
SF_REFRESH_TOKEN         # optional — OAuth 2.0 refresh token for that app

# ── Quick Start ──────────────────────────────────────────────────────────────
# 1. cp .env.example .env  (and fill in your API keys)
# 2. npm install
# 3. npm start
